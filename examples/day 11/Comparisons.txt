# Day 11 - Arduino vs Raspberry Pi Pico: 4-Key Membrane Keypad Comparisons

This file compares Day 11 keypad activities on Raspberry Pi Pico (MicroPython) and Arduino MKR Zero (C++/Arduino language).

---

## Hardware Setup

| Feature | Raspberry Pi Pico | Arduino MKR Zero |
|---|---|---|
| Keypad type | 4-key membrane (1 row + 4 columns) | 4-key membrane (1 row + 4 columns) |
| Keypad common row | 3.3V | 3.3V |
| Key input pins | GPIO10, GPIO11, GPIO12, GPIO13 | D10, D11, D12, D13 |
| Key mapping from lesson | key1->GPIO11, key2->GPIO10, key3->GPIO13, key4->GPIO12 | key1->D11, key2->D10, key3->D13, key4->D12 |
| Block LED pin | GPIO6 | D6 |
| Strand data pin | GPIO2 | D2 |
| Strand LED count | 15 | 15 |

---

## Activity 1: Basic Keypad Test

### Raspberry Pi Pico (MicroPython)

```python
from machine import Pin
import time

key1 = Pin(11, Pin.IN, Pin.PULL_DOWN)
key2 = Pin(10, Pin.IN, Pin.PULL_DOWN)
key3 = Pin(13, Pin.IN, Pin.PULL_DOWN)
key4 = Pin(12, Pin.IN, Pin.PULL_DOWN)

while True:
	if key1.value() == 1:
		print("Button 1")
	if key2.value() == 1:
		print("Button 2")
	if key3.value() == 1:
		print("Button 3")
	if key4.value() == 1:
		print("Button 4")
	time.sleep(0.3)
```

### Arduino MKR Zero (C++)

```cpp
const int key1 = 11;
const int key2 = 10;
const int key3 = 13;
const int key4 = 12;

void setup() {
	Serial.begin(9600);
	pinMode(key1, INPUT_PULLDOWN);
	pinMode(key2, INPUT_PULLDOWN);
	pinMode(key3, INPUT_PULLDOWN);
	pinMode(key4, INPUT_PULLDOWN);
}

void loop() {
	if (digitalRead(key1) == HIGH) {
		Serial.println("Button 1");
	}
	if (digitalRead(key2) == HIGH) {
		Serial.println("Button 2");
	}
	if (digitalRead(key3) == HIGH) {
		Serial.println("Button 3");
	}
	if (digitalRead(key4) == HIGH) {
		Serial.println("Button 4");
	}
	delay(300);
}
```

## Activity 2: LED Strand Colour Selector (Function Arguments)

### Raspberry Pi Pico (MicroPython)

```python
from machine import Pin
import time
from neopixel import NeoPixel

key1 = Pin(11, Pin.IN, Pin.PULL_DOWN)
key2 = Pin(10, Pin.IN, Pin.PULL_DOWN)
key3 = Pin(13, Pin.IN, Pin.PULL_DOWN)
key4 = Pin(12, Pin.IN, Pin.PULL_DOWN)

strand = NeoPixel(Pin(2), 15)
ledindex = list(range(15))

off = (0, 0, 0)
green = (0, 255, 0)
blue = (0, 0, 255)
white = (255, 255, 255)
yellow = (255, 255, 0)
pink = (255, 0, 255)
aqua = (0, 255, 255)

def blinky1(colour1, colour2):
	for led in ledindex:
		if (led % 2) == 0:
			strand[led] = colour1
		else:
			strand[led] = colour2
	strand.write()

strand.fill((0, 0, 0))
strand.write()
time.sleep(1)

while True:
	time.sleep(0.1)

	if key1.value() == 1:
		blinky1(aqua, green)
	elif key2.value() == 1:
		blinky1(white, blue)
	elif key3.value() == 1:
		blinky1(yellow, pink)
	elif key4.value() == 1:
		blinky1(off, off)
```

### Arduino MKR Zero (C++)

```cpp
#include <Adafruit_NeoPixel.h>

const int key1 = 11;
const int key2 = 10;
const int key3 = 13;
const int key4 = 12;

Adafruit_NeoPixel strand(15, 2, NEO_RGB + NEO_KHZ800);

uint32_t offc;
uint32_t green;
uint32_t blue;
uint32_t white;
uint32_t yellow;
uint32_t pink;
uint32_t aqua;

void blinky1(uint32_t colour1, uint32_t colour2) {
	for (int led = 0; led < 15; led++) {
		if ((led % 2) == 0) {
			strand.setPixelColor(led, colour1);
		} else {
			strand.setPixelColor(led, colour2);
		}
	}
	strand.show();
}

void setup() {
	pinMode(key1, INPUT_PULLDOWN);
	pinMode(key2, INPUT_PULLDOWN);
	pinMode(key3, INPUT_PULLDOWN);
	pinMode(key4, INPUT_PULLDOWN);

	strand.begin();
	strand.clear();
	strand.show();

	offc = strand.Color(0, 0, 0);
	green = strand.Color(0, 255, 0);
	blue = strand.Color(0, 0, 255);
	white = strand.Color(255, 255, 255);
	yellow = strand.Color(255, 255, 0);
	pink = strand.Color(255, 0, 255);
	aqua = strand.Color(0, 255, 255);

	delay(1000);
}

void loop() {
	delay(100);

	if (digitalRead(key1) == HIGH) {
		blinky1(aqua, green);
	} else if (digitalRead(key2) == HIGH) {
		blinky1(white, blue);
	} else if (digitalRead(key3) == HIGH) {
		blinky1(yellow, pink);
	} else if (digitalRead(key4) == HIGH) {
		blinky1(offc, offc);
	}
}
```

### Key Differences

| Concept | MicroPython (Pico) | Arduino (C++) |
|---|---|---|
| Function arguments | `def blinky1(colour1, colour2)` | `void blinky1(uint32_t colour1, uint32_t colour2)` |
| Colour type | Tuple `(r,g,b)` | Encoded `uint32_t` from `Color()` |

## Activity 3: Push to Flash (Hold Key)

### Raspberry Pi Pico (MicroPython)

```python
from machine import Pin
import time

key1 = Pin(11, Pin.IN, Pin.PULL_DOWN)
key2 = Pin(10, Pin.IN, Pin.PULL_DOWN)
key3 = Pin(13, Pin.IN, Pin.PULL_DOWN)
key4 = Pin(12, Pin.IN, Pin.PULL_DOWN)

blockLED = Pin(6, Pin.OUT)
flash = 0

def program1():
	blockLED.value(1)
	time.sleep(flash)
	blockLED.value(0)
	time.sleep(flash)

while True:
	while key1.value() == 1:
		flash = 1
		program1()

	while key2.value() == 1:
		flash = 0.5
		program1()

	while key3.value() == 1:
		flash = 0.1
		program1()

	while key4.value() == 1:
		flash = 0.05
		program1()
```

### Arduino MKR Zero (C++)

```cpp
const int key1 = 11;
const int key2 = 10;
const int key3 = 13;
const int key4 = 12;
const int blockLED = 6;

float flash = 0.0;

void program1() {
	digitalWrite(blockLED, HIGH);
	delay((int)(flash * 1000));
	digitalWrite(blockLED, LOW);
	delay((int)(flash * 1000));
}

void setup() {
	pinMode(key1, INPUT_PULLDOWN);
	pinMode(key2, INPUT_PULLDOWN);
	pinMode(key3, INPUT_PULLDOWN);
	pinMode(key4, INPUT_PULLDOWN);
	pinMode(blockLED, OUTPUT);
}

void loop() {
	while (digitalRead(key1) == HIGH) {
		flash = 1.0;
		program1();
	}
	while (digitalRead(key2) == HIGH) {
		flash = 0.5;
		program1();
	}
	while (digitalRead(key3) == HIGH) {
		flash = 0.1;
		program1();
	}
	while (digitalRead(key4) == HIGH) {
		flash = 0.05;
		program1();
	}
}
```

## Activity 4: Block Duplicate Inputs (Wait for All LOW)

### Raspberry Pi Pico (MicroPython)

```python
from machine import Pin
import time

key1 = Pin(11, Pin.IN, Pin.PULL_DOWN)
key2 = Pin(10, Pin.IN, Pin.PULL_DOWN)
key3 = Pin(13, Pin.IN, Pin.PULL_DOWN)
key4 = Pin(12, Pin.IN, Pin.PULL_DOWN)

state = 0

while True:
	time.sleep(0.1)

	if state == 0:
		if key1.value() == 1:
			print("Button 1")
			state = 1
		elif key2.value() == 1:
			print("Button 2")
			state = 1
		elif key3.value() == 1:
			print("Button 3")
			state = 1
		elif key4.value() == 1:
			print("Button 4")
			state = 1

	elif state == 1 and key1.value() == key2.value() == key3.value() == key4.value() == 0:
		state = 0
```

### Arduino MKR Zero (C++)

```cpp
const int key1 = 11;
const int key2 = 10;
const int key3 = 13;
const int key4 = 12;
int state = 0;

void setup() {
	Serial.begin(9600);
	pinMode(key1, INPUT_PULLDOWN);
	pinMode(key2, INPUT_PULLDOWN);
	pinMode(key3, INPUT_PULLDOWN);
	pinMode(key4, INPUT_PULLDOWN);
}

void loop() {
	delay(100);

	if (state == 0) {
		if (digitalRead(key1) == HIGH) {
			Serial.println("Button 1");
			state = 1;
		} else if (digitalRead(key2) == HIGH) {
			Serial.println("Button 2");
			state = 1;
		} else if (digitalRead(key3) == HIGH) {
			Serial.println("Button 3");
			state = 1;
		} else if (digitalRead(key4) == HIGH) {
			Serial.println("Button 4");
			state = 1;
		}
	} else if (
			state == 1 &&
			digitalRead(key1) == LOW &&
			digitalRead(key2) == LOW &&
			digitalRead(key3) == LOW &&
			digitalRead(key4) == LOW) {
		state = 0;
	}
}
```

## Activity 5: Secret Present List Passcode

### Raspberry Pi Pico (MicroPython)

```python
from machine import Pin
import time

key1 = Pin(11, Pin.IN, Pin.PULL_DOWN)
key2 = Pin(10, Pin.IN, Pin.PULL_DOWN)
key3 = Pin(13, Pin.IN, Pin.PULL_DOWN)
key4 = Pin(12, Pin.IN, Pin.PULL_DOWN)
blockLED = Pin(6, Pin.OUT)

presents = ["Train set", "Furby", "Boomerang", "YoYo"]
passcode = [1, 2, 3, 4]
userentry = []
state = 0
key = 0

def appendkey():
	global state
	userentry.append(key)
	print("*", end="")
	state = 1

def myprint(mytext):
	print(mytext)
	time.sleep(0.5)

print("")
print("Welcome to the Secret Present List system")
time.sleep(1)
print("Enter the passcode to continue: ", end="")

while len(userentry) < 4:
	time.sleep(0.1)

	if state == 0:
		if key1.value() == 1:
			key = 1
			appendkey()
		elif key2.value() == 1:
			key = 2
			appendkey()
		elif key3.value() == 1:
			key = 3
			appendkey()
		elif key4.value() == 1:
			key = 4
			appendkey()

	elif state == 1 and key1.value() == key2.value() == key3.value() == key4.value() == 0:
		state = 0

if userentry == passcode:
	blockLED.value(1)
	print("\n")
	print("----------------------")
	print("*** ACCESS GRANTED ***")
	myprint("----------------------")
	myprint("Secret present list:")
	for i in presents:
		myprint(i)
	myprint("----------------------")
	blockLED.value(0)
else:
	myprint("\n------------------")
	myprint("INCORRECT PASSCODE")
	myprint("ACCESS DENIED")
	time.sleep(1)
```

### Arduino MKR Zero (C++)

```cpp
const int key1 = 11;
const int key2 = 10;
const int key3 = 13;
const int key4 = 12;
const int blockLED = 6;

int passcode[4] = {1, 2, 3, 4};
int userentry[4] = {0, 0, 0, 0};
int entryIndex = 0;
int state = 0;
int key = 0;

const char* presents[4] = {"Train set", "Furby", "Boomerang", "YoYo"};

void appendKey() {
	userentry[entryIndex] = key;
	entryIndex++;
	Serial.print("*");
	state = 1;
}

void myprint(const char* text) {
	Serial.println(text);
	delay(500);
}

bool codeMatches() {
	for (int i = 0; i < 4; i++) {
		if (userentry[i] != passcode[i]) return false;
	}
	return true;
}

void setup() {
	Serial.begin(9600);
	pinMode(key1, INPUT_PULLDOWN);
	pinMode(key2, INPUT_PULLDOWN);
	pinMode(key3, INPUT_PULLDOWN);
	pinMode(key4, INPUT_PULLDOWN);
	pinMode(blockLED, OUTPUT);

	Serial.println();
	Serial.println("Welcome to the Secret Present List system");
	delay(1000);
	Serial.print("Enter the passcode to continue: ");

	while (entryIndex < 4) {
		delay(100);

		if (state == 0) {
			if (digitalRead(key1) == HIGH) {
				key = 1;
				appendKey();
			} else if (digitalRead(key2) == HIGH) {
				key = 2;
				appendKey();
			} else if (digitalRead(key3) == HIGH) {
				key = 3;
				appendKey();
			} else if (digitalRead(key4) == HIGH) {
				key = 4;
				appendKey();
			}
		} else if (
				state == 1 &&
				digitalRead(key1) == LOW &&
				digitalRead(key2) == LOW &&
				digitalRead(key3) == LOW &&
				digitalRead(key4) == LOW) {
			state = 0;
		}
	}

	if (codeMatches()) {
		digitalWrite(blockLED, HIGH);
		Serial.println("\n");
		Serial.println("----------------------");
		Serial.println("*** ACCESS GRANTED ***");
		myprint("----------------------");
		myprint("Secret present list:");
		for (int i = 0; i < 4; i++) {
			myprint(presents[i]);
		}
		myprint("----------------------");
		digitalWrite(blockLED, LOW);
	} else {
		myprint("\n------------------");
		myprint("INCORRECT PASSCODE");
		myprint("ACCESS DENIED");
		delay(1000);
	}
}

void loop() {
}
```

## General Language Comparison: Day 11 Concepts

| Concept | MicroPython (Pico) | Arduino (C++) |
|---|---|---|
| Key input setup | `Pin(n, Pin.IN, Pin.PULL_DOWN)` | `pinMode(n, INPUT_PULLDOWN)` |
| Read key | `key.value()` | `digitalRead(key)` |
| Debounce/state lock | Integer `state` + all-LOW check | Same logic with `state` and LOW checks |
| Function args for colors | Tuples passed as args | `uint32_t` color values passed as args |
| Append input | `list.append(value)` | Array + index increment |
| Length check | `len(userentry)` | `entryIndex < targetLength` |
| Print same line | `print("*", end="")` | `Serial.print("*")` |
| LED on/off | `blockLED.value(1/0)` | `digitalWrite(blockLED, HIGH/LOW)` |
